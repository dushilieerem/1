简单动量与复合动量（马睿祥）

~~~python
def momentum_5d(self, window: int = 5):
    """5 日动量：过去 window 日收益率。"""
    return self.close.pct_change(window)
        
def momentum_ratio(self, short: int = 5, long: int = 20):
    """动量比：短期动量 / 长期动量。"""
    short_ret = self.close.pct_change(short)
    long_ret = self.close.pct_change(long).abs()
    return short_ret / (long_ret + 1e-10)

def momentum_acceleration(self, fast: int = 5, slow: int = 20):
    """动量加速度：短期动量 - 长期动量。"""
    return self.close.pct_change(fast) - self.close.pct_change(slow)
    
def momentum_decay(self, window: int = 20, decay: float = 0.9):
    """指数衰减动量：对每日收益率加权平均，越近权重越高。"""
    ret = self.close.pct_change()
    weights = [decay ** i for i in range(window)]
    weights = weights[::-1]  # 最近权重最大
    weights = [w / sum(weights) for w in weights]
    return ret.rolling(window).apply(lambda x: (x * weights).sum(), raw=True)
    
def momentum_risk_adjusted(self, window: int = 20):
    """风险调整动量：收益率 / 波动率。"""
    ret = self.close.pct_change(window)
    vol = self.close.pct_change().rolling(window).std()
    return ret / (vol + 1e-10)
~~~

反转与趋势与情绪类（赵文博）

~~~python
def reversal_1d(self):
    """1 日反转。"""
    return -self.close.pct_change(1)

def trend_strength(self, window: int = 20):
    """趋势强度：当前价 / 均价。"""
    return self.close / self.close.rolling(window).mean()

def trend_position(self, window: int = 60):
    """趋势位置：（当前价 - 最低价）/（最高价 - 最低价）。"""
    hh = self.high.rolling(window).max()
    ll = self.low.rolling(window).min()
    return (self.close - ll) / (hh - ll + 1e-10)

def trend_consistency(self, window: int = 10):
    """趋势一致性：上涨天数占比。"""
    up = (self.close.diff() > 0).astype(float)
    return up.rolling(window).mean()

def ar(self):
    """聚宽 AR：26 日人气指标。"""
    return (
        (self.high - self.open).rolling(26).sum()
        / (self.open - self.low).rolling(26).sum()
        * 100
    )
~~~

情绪类（于旭东）

~~~python
def _ar(self):
    return (
        (self.high - self.open).rolling(26).sum()
        / (self.open - self.low).rolling(26).sum()
        * 100
    )

def _br(self):
    previous_close = self.close.shift()
    up = (self.high - previous_close).clip(lower=0)
    down = (previous_close - self.low).clip(lower=0)
    return up.rolling(26).sum() / down.rolling(26).sum() * 100

def _true_range(self):
    previous_close = self.close.shift()
    first = self.high - self.low
    second = (self.high - previous_close).abs()
    third = (self.low - previous_close).abs()
    result = first.where(first >= second, second)
    return result.where(result >= third, third)

def _turnover_mean(self, window: int):
    return self.turnover.rolling(window).mean()

def _wvad(self):
    value = (self.close - self.open) / (self.high - self.low) * self.volume
    return value.rolling(6).sum()

def arbr(self):
    """聚宽 ARBR：AR 与 BR 的差。"""
    return self._ar() - self._br()

def atr6(self):
    """聚宽 ATR6：6 日真实振幅均值。"""
    return self._true_range().rolling(6).mean()

def br(self):
    """聚宽 BR：26 日意愿指标。"""
    return self._br()

def davol5(self):
    """聚宽 DAVOL5：5 日与 120 日平均换手率之比。"""
    return self._turnover_mean(5) / self._turnover_mean(120)

def mawvad(self):
    """聚宽 MAWVAD：WVAD 的 6 日移动均值。"""
    return self._wvad().rolling(6).mean()
~~~

情绪类（王为敏）

~~~python
def money_flow_20(self):
    """聚宽 money_flow_20：20 日典型价格资金流量。"""
    typical_price = (self.high + self.low + self.close) / 3
    return (typical_price * self.volume).rolling(20).sum()

def psy(self):
    """聚宽 PSY：12 日上涨天数占比。"""
    return (self.close.diff() > 0).rolling(12).mean() * 100

def turnover_volatility(self):
    """聚宽 turnover_volatility：20 日换手率标准差。"""
    return self.turnover.rolling(20).std()

def tvma6(self):
    """聚宽 TVMA6：6 日成交金额移动均值。"""
    return self.amount.rolling(6).mean()
    
def _vema(self, window: int):
    return self.volume.ewm(span=window, adjust=False).mean()    
    
def _vdiff(self):
    return self._vema(12) - self._vema(26)
    
def vdea(self):
    """聚宽 VDEA：VDIFF 的 9 日指数移动均线。"""
    return self._vdiff().ewm(span=9, adjust=False).mean()
~~~
